#include <stdio.h>

int main() {
    int voto;
    int somma = 0;
    int contatore = 0;

    printf("Inserisci voto: ");
    scanf("%d", &voto);

    while(voto != -1) {
        if(voto < 1 || voto > 10) {
            printf("Voto non valido: ");
            scanf("%d", &voto);
        } else {
            somma = somma + voto;
            contatore++;
            printf("Inserisci voto: ");
            scanf("%d", &voto);
        }
    }

    if (contatore > 0) {
        float media = (float)somma / contatore;
        printf("Media: %f\n", media);
    } else {
        printf("Nessun voto inserito.\n");
    }

    return 0;
}
