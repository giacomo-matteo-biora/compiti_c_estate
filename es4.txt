#include <stdio.h>

int main() {
    int n=0;
    int i=0;
    float temp=0.0;
    int febbre = 0;
    int scartati = 0;

    printf("Inserisci numero pazienti: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++) {
        printf("Inserisci temperatura: ");
        scanf("%f", &temp);

        if(temp < 34.0 || temp > 40.0) {
            scartati++;
        } else if(temp > 37.5) {
            febbre++;
        }
    }

    printf("Pazienti con febbre: %d\n", febbre);
    printf("Misurazioni scartate: %d\n", scartati);

    return 0;
}
