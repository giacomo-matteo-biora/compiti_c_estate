/** Jack Biora **/

#include <stdio.h>

int main() {
    float litri=0.0;
    int pos =0;
    int b = 0;
    float imp= 0.0;
    int giorno=0;
    int b_doppio = 0;
    float tot_incasso = 0.0;

    printf("Litri: ");
    scanf("%f", &litri);

    while (litri != -1) {

	printf("Giorno 1-7: ");
        scanf("%d", &giorno);        

	printf("Importo: ");
        scanf("%f", &imp);
        

        pos++;
        tot_incasso = tot_incasso + imp;

        if ((litri >= 30 && imp >= 40) || (giorno >= 1 && giorno <= 5 && imp >= 60)) {
            b++;
            if (pos % 10 == 0) {
                b_doppio++;
            }
        }

        printf("Litri: ");
        scanf("%f", &litri);
    }

    printf("Bonus: %d\nBonus doppio: %d\nIncasso: %f euro\n", b, b_doppio, tot_incasso);

    return 0;
}
