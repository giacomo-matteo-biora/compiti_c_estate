#include <stdio.h>

int main() {
    int n, i;
    float peso=0.0;
    int conformi_cont =0;
    leggeri_cont = 0;
    pesanti_cont = 0;
    int scarti_totali;

    printf("Inserisci numero pezzo: ");
    scanf("%d", &n);

    while (n <= 0) {
        printf("Numero non valido: ");
        scanf("%d", &n);
    }

    for (i = 0; i < n; i++) {
        printf("Inserisci peso pezzo: ");
        scanf("%f", &peso);

        while (peso <= 0) {
            printf("Peso non validi: ");
            scanf("%f", &peso);
        }

        if (peso >= 95 && peso <= 105) {
            conformi_cont++;
        } else if (peso < 95) {
            leggeri_cont++;
        } else {
            pesanti_cont++;
        }
    }

    scarti_totali = leggeri_cont + pesanti_cont;

    printf("Prezzi conformi: %d\n", conformi_cont);
    printf("Pezzi leggeri: %d\n", leggeri_cont);
    printf("Pezzi pesanti: %d\n", pesanti_cont);

    return 0;
}
