#include <stdio.h>

int main() {
    float temp= 0.0;
    float press = 0.0; 
    float liv =0.0;
    int allarmi = 0;
    int cons = 0;
    int max_cons= 0;

    printf("Temperatura: ");
    scanf("%f", &temp);

    while (temp != -999) {
        printf("Pressione: ");
        scanf("%f", &press);

        printf("Livello serbatio : ");
        scanf("%f", &liv);

        while (liv < 0 || liv > 100) {
            printf("Livello invaliod: ");
            scanf("%f", &liv);
        }

        if ((temp > 80 && press > 5) || liv < 10 || temp > 120) {
            allarmi++;
            cons++;
            if (cons > max_cons) {
                max_cons = cons;
            }
        } else {
            cons = 0;
        }

        printf("Temperatura ");
        scanf("%f", &temp);
    }

    printf("Tot allarmi: %d\n", allarmi);
    printf("Consecutivi: %d\n", max_cons);

    return 0;
}
