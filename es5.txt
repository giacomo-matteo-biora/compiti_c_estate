#include <stdio.h>

int main() {
    
    int n, i, eta;
    int maggiorenni_contatore = 0;
    int minorenni_contatore = 0;
    int non_validi_contatore = 0;

    printf("Inserisci il numero di visitatori: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++) {
        printf("Inserisci eta: ");
        scanf("%d", &eta);

        if (eta < 1 || eta > 99) {
            printf("Eta non valida!\n");
            non_validi_contatore++;
        } else if (eta >= 18) {
            maggiorenni_contatore++;
        } else {
            minorenni_contatore++;
        }
    }

    printf("Maggiorenni: %d\n", maggiorenni_contatore);
    printf("Minorenni: %d\n", minorenni_contatore);
    printf("Eta non valide: %d\n", non_validi_contatore);

    return 0;
}
